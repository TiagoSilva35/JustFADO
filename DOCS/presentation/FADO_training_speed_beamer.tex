% ---- 01 the change ----
\pageset{1.The change}{01}
\begin{frame}
  \ctitle{The change: an analytic fairness gradient}
  \tb{11.7}{0.85}{2.2}{\color{body}\fs{15}{20}Each online step ran \textbf{two} backward passes: one for the task loss, and a second (on a persistent tape, once per sample) only to read $\partial n/\partial W$, $\partial n/\partial b$ for the fairness penalty.}
  \panel{11.3}{0.85}{3.1}{\fs{14}{18.5}%
    {\color{cred}\bfseries Why the second pass was never needed.}\par
    \vspace{2pt}
    {\color{body}The fairness penalty is a function of the \emph{node activations}, so its gradient is the node's own \textbf{local} Jacobian --- there is no upstream chain to assemble. With $n=\sigma(z)$, $z=(W^{\top}x+b)/\tau$:}\par
    \vspace{4pt}
    \centering$\displaystyle \frac{\partial n}{\partial W}=x\,\sigma'(z)^{\top}/\tau,\qquad
      \frac{\partial n}{\partial b}=\sigma'(z)/\tau,\qquad \sigma'(z)=\sigma(z)\bigl(1-\sigma(z)\bigr).$\par
    \vspace{4pt}
    \raggedright{\color{body}And $\sigma(z)$ \emph{is} the forward pass's {\monofont node\_decisions} --- already computed. One elementwise product, one outer product, per tree. No tape.}}
  \tb{11.6}{0.85}{5.55}{\color{ink}\fs{13.5}{18}\setlength{\parskip}{5pt}
    {\color{cred}\textbullet}\ The \textbf{task loss keeps autodiff}: its target is the prediction, so it genuinely needs the full upstream chain (leaf mixture, gates, $\theta$).\par
    {\color{cred}\textbullet}\ Applied to all three training loops \textperiodcentered{} the tape is no longer persistent \textperiodcentered{} {\monofont 'leaf'} mode still uses autodiff.}
  \hrl{0.85}{6.75}{11.6}
  \tb{11.6}{0.85}{6.9}{\color{body}\fs{12.5}{15}Autodiff is worth its cost only when you need the \emph{upstream} chain. The fairness term never leaves the top of the graph.}
\end{frame}

% ---- 02 verified ----
\pageset{2.Verified}{02}
\begin{frame}
  \ctitle{Verified: bit-identical}
  \tb{5.75}{0.85}{2.35}{\bfseries\color{ink}\fs{16}{19}Unit: both paths, same state}
  \tb{5.75}{0.85}{2.85}{\color{body}\fs{13.5}{18}\setlength{\parskip}{5pt}
    {\monofont gradient\_w}, {\monofont gradient\_b}, {\monofont agg\_y} all differ by {\color{green}\textbf{0.00e+00}}; counts match.\par
    Across $\tau \in \{1.0,\ 0.35,\ 0.1\}$ and both update rules (vanilla, EMA).}
  \tb{5.75}{7.05}{2.35}{\bfseries\color{ink}\fs{16}{19}End-to-end A/B}
  \tb{5.75}{7.05}{2.85}{\color{body}\fs{13.5}{18}\setlength{\parskip}{5pt}
    Original code restored and re-run on the same seed, 400 online steps.\par
    DP, EO and accuracy trajectories \textbf{and} all 12 final weight tensors: {\color{green}\textbf{identical}}.}
  \hrl{0.85}{4.5}{11.6}
  \panel{11.3}{0.85}{4.75}{\fs{14}{19}%
    {\color{cred}\bfseries Why exactly zero, and not just close?}\par
    \vspace{2pt}
    {\color{body}Because the analytic path \textbf{reuses the forward's $\sigma(z)$} instead of recomputing $z$, there is not even float32 rounding drift. (Recomputing $z$ gives $\sim$1e-6; reusing it gives 0.)\par
    \vspace{3pt}
    Consequence: \textbf{the published numbers cannot move.} This is a pure speed change.}}
  \tb{11.6}{0.85}{6.6}{\color{body}\fs{12.5}{15}Bonus: trees are now indexed directly instead of inferred from gradient shapes --- immune to a $\theta$ whose leading dimension equals {\monofont data\_dim}.}
\end{frame}

% ---- 04 next ----
\pageset{4.Next}{04}
\begin{frame}
  \ctitle{Next: the actual bottleneck}
  \tb{11.7}{0.85}{2.3}{\color{body}\fs{15.5}{21}{\monofont compute\_fairness\_gradients} operates on state that is \textbf{already NumPy} ({\monofont gradient\_w}, {\monofont gradient\_b} are plain arrays) --- but rebuilds every term with Python {\monofont sum()} comprehensions and per-element {\monofont tf.convert\_to\_tensor}, once per gradient tensor, every step.}
  \panel{11.3}{0.85}{3.5}{\fs{14.5}{19}%
    {\color{cred}\bfseries $\approx$196{,}000 tensor constructions per 400 steps} --- to do arithmetic on arrays of a few dozen numbers.\par
    \vspace{3pt}
    {\color{body}Fix: rewrite it in pure NumPy. No TF in the loop, no wrapping, no host syncs. Same treatment as today: prove bit-identical, then A/B the full trajectory.}}
  \tb{11.6}{0.85}{5.1}{\color{ink}\fs{14}{19}\setlength{\parskip}{5pt}
    {\color{cred}\textbullet}\ Then: \textbf{incremental fairness window} --- $O(1)$ running counters instead of recomputing DP/EO over the whole window every step.\par
    {\color{cred}\textbullet}\ Then: \textbf{graph the forward} with {\monofont @tf.function} (eager today) \textperiodcentered{} batch the trees instead of a Python loop.}
  \hrl{0.85}{6.4}{11.6}
  \tb{11.6}{0.85}{6.6}{\color{body}\fs{13}{16}Lesson worth keeping: the derivation was right and the component got 540$\times$ faster --- but it was 7\% of the step. \textbf{Profile first, then optimise.}}
\end{frame}
