% ---- 2.1 CapyMOA (05) ----
\pageset{2.Background}{05}
\begin{frame}
  \ctitle{CapyMOA}
  \tb{11.7}{0.85}{2.3}{\color{body}\fs{15}{20}A Python library for machine learning on data streams, built on MOA (Java) with a PyTorch bridge. It provides streaming classifiers (e.g.\ ARF), drift detectors (ADWIN), prequential evaluation, dataset loaders, and drift generators.}
  \hrl{0.85}{3.55}{11.6}
  \tb{5.8}{0.85}{3.75}{\bfseries\color{ink}\fs{15}{18}Does it improve performance?}
  \tb{5.8}{0.85}{4.25}{\color{body}\fs{13}{16.5}\setlength{\parskip}{5pt}
    {\color{cred}Base learner: no.} It does not contain our forest, so it cannot make the model itself more accurate.\par
    {\color{green}Detector: yes --- it is swappable.} The controller reacts to a drift signal, so we can replace ADWIN with a better detector (SEED, STEPD, DDM, HDDM, \ldots) and benchmark it.\par
    {\color{green}Baselines: yes.} Standard, tested, efficient ARF and streaming trees.}
  \tb{5.8}{7.05}{3.75}{\bfseries\color{ink}\fs{15}{18}Can it make deeper trees affordable?}
  \tb{5.8}{7.05}{4.25}{\color{body}\fs{13}{16.5}\setlength{\parskip}{5pt}
    The real cost: FADO's forest is our own implementation, so more depth means a lot more training time --- that is what makes sweeping tree height expensive today.\par
    {\color{cred}CapyMOA does not run our forest, so it cannot speed it up} --- its streaming learners are different algorithms.\par
    To afford deeper trees the lever is our own code: {\color{ink}vectorise} across trees / nodes (depth is parallel width) and parallelise the sweep. GPUs help only if we batch updates over a window --- one-at-a-time online steps are too small to fill one.}
\end{frame}

% ---- 2.2 speeding up training (06) ----
\pageset{2.Background}{06}
\begin{frame}
  \ctitle{Speeding up our own training}
  \tb{11.7}{0.85}{2.2}{\color{body}\fs{15}{20}The forest is small --- almost all the per-step cost is overhead, not math. These changes cut it, and \textbf{none change our results}.}
  \panel{11.3}{0.85}{3.15}{\fs{14}{18.5}%
    {\color{cred}\bfseries Main upgrade --- analytic fairness gradient.}\par
    \vspace{2pt}
    {\color{body}Every step now runs a \emph{second} backward pass (a persistent tape) just to read $\partial n/\partial W$ and $\partial n/\partial b$ for the fairness penalty. But a node is $n=\sigma\!\big((w^{\top}x+b)/\tau\big)$, so these are closed-form:}\par
    \vspace{4pt}
    \centering$\displaystyle \frac{\partial n}{\partial W}=x\,\sigma'(z)^{\top}/\tau,\qquad
      \frac{\partial n}{\partial b}=\sigma'(z)/\tau,\qquad z=(w^{\top}x+b)/\tau.$\par
    \vspace{4pt}
    \raggedright{\color{body}Replacing the extra autodiff pass with this formula drops about \textbf{half the backward work} per step --- identical numbers, confirmed by a gradient check.}}
  \tb{11.6}{0.85}{5.75}{\color{ink}\fs{13.5}{18}\setlength{\parskip}{5pt}
    {\color{cred}\textbullet}\ \textbf{Pure-NumPy fairness math} --- the state is already NumPy; drop the per-step {\monofont tf.convert\_to\_tensor}.\par
    {\color{cred}\textbullet}\ \textbf{Incremental fairness window} --- $O(1)$ running counters instead of recomputing the whole window each step.\par
    {\color{cred}\textbullet}\ \textbf{Graph the forward} --- wrap forward + loss in {\monofont @tf.function} (eager today) \textperiodcentered{} batch the trees instead of a Python loop.}
  \hrl{0.85}{6.8}{11.6}
  \tb{11.6}{0.85}{6.95}{\color{body}\fs{12.5}{15}All speed-only --- this is what makes sweeping tree depth and the 1M-row BAF runs affordable.}
\end{frame}

% ---- 2.3 drift generation (07) ----
\pageset{2.Background}{07}
\begin{frame}
  \ctitle{Drift generation (KDD'26)}
  \tb{11.7}{0.85}{2.3}{\color{body}\fs{15}{20}A method (Cerqueira et al., KDD'26 --- same group as CapyMOA) to inject controlled, labelled concept drift into a real data stream: shuffle the stream, pick a random onset point, apply a typed change, either abrupt or gradual.}
  \tb{6.5}{0.85}{3.55}{\color{ink}\fs{14.5}{21}\setlength{\parskip}{5pt}
    {\color{cred}\textbullet}\ \textbf{Class-prior} --- drop a class with some probability\par
    {\color{cred}\textbullet}\ \textbf{Class-label swap} --- relabel one class to another\par
    {\color{cred}\textbullet}\ \textbf{Feature permutation} --- reorder the features\par
    {\color{cred}\textbullet}\ \textbf{Feature filtering} --- drop rows above a threshold}
  \tb{4.9}{7.6}{3.55}{\color{body}\fs{13.5}{18}\setlength{\parskip}{6pt}
    Our COMPAS drift is already a class-label swap --- adopting this makes it standard and citable.\par
    Note: it is drift/accuracy focused; its own datasets have no protected attributes.}
\end{frame}

% ---- 2.4 per-node lambda (08) ----
\pageset{2.Background}{09}
\begin{frame}
  \ctitle{Per-node / per-tree $\lambda$}
  \tb{11.7}{0.85}{2.4}{\color{body}\fs{16}{22}\setlength{\parskip}{9pt}
    Today the fairness weight $\lambda$ in $\mathcal{L}=\mathcal{L}_{\mathrm{CE}}+\lambda\,\mathcal{L}_{\mathrm{fair}}$ is a single global constant.\par
    Idea: let $\lambda$ vary \textbf{per node or per tree} --- larger where a node's group disparity is larger, smaller where fairness is already met. Each node already tracks group statistics, so it fits naturally.\par
    Goal: a better accuracy / fairness trade-off. The benefit is uncertain and needs an ablation to confirm it beats a global $\lambda$.}
\end{frame}

% ---- 3.1 combining (10) ----
\pageset{3.Plan}{11}
\begin{frame}
  \ctitle{Combining drift generation with fairness data}
  \tb{11.7}{0.85}{2.4}{\color{body}\fs{16}{22}The drift framework is accuracy-focused; BAF and Folktables carry fairness. The two can be combined.}
  \tb{11.2}{0.9}{3.5}{\color{ink}\fs{15.5}{21}\setlength{\parskip}{9pt}
    Inject a typed, labelled drift (from the KDD'26 method) onto a \textbf{protected subgroup} of a fairness dataset --- for example, a class-label swap that affects one group only.\par
    This gives controlled fairness-under-drift scenarios with \textbf{known onset points}: we can measure when fairness breaks and how the controller responds.}
  \hrl{0.85}{5.75}{11.6}
  \tb{11.6}{0.85}{5.95}{\color{body}\fs{14}{18}This combines the drift generator, a real dataset, and the controller into one experiment that existing benchmarks do not provide.}
\end{frame}
